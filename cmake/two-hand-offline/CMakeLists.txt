cmake_minimum_required(VERSION 3.27)
project(sm2_twohand_offline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR _ITERATOR_DEBUG_LEVEL=0)
add_compile_options(/EHsc /Zc:__cplusplus /Zc:preprocessor)
get_filename_component(R "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(SM2VR_OFFLINE_DEPS "${R}/build/mod/_deps" CACHE PATH "Existing dependency sources; no downloads")
set(D "${SM2VR_OFFLINE_DEPS}")
foreach(dep spdlog minhook)
 if(NOT EXISTS "${D}/${dep}-src/CMakeLists.txt")
  message(FATAL_ERROR "Existing ${dep} source required under SM2VR_OFFLINE_DEPS; this target does not download or install")
 endif()
endforeach()
find_path(SM2VR_OPENVR_HEADERS openvr.h HINTS "${R}/refs/vrframework-core/extern/openvr/headers" "${D}/../../../refs/vrframework-core/extern/openvr/headers" NO_DEFAULT_PATH REQUIRED)
set(SPDLOG_BUILD_SHARED OFF)
add_subdirectory("${D}/spdlog-src" spdlog)
add_subdirectory("${D}/minhook-src" minhook)
enable_testing()
function(test name)
 add_executable(${name}_tests ${ARGN})
 target_include_directories(${name}_tests PRIVATE "${R}/mod/src" "${R}/tests" "${SM2VR_OPENVR_HEADERS}")
 if(NOT name MATCHES "^(weapon_aim|engine_arm_driver|vr_session)$")
  add_test(NAME ${name} COMMAND ${name}_tests)
 endif()
endfunction()
test(two_hand_aim "${R}/tests/test_two_hand_aim.cpp")
test(weapon_aim "${R}/tests/test_weapon_aim.cpp")
add_test(NAME weapon_aim_config COMMAND weapon_aim_tests "${R}/config/index_weapon.calibration")
test(vr_combat_routing "${R}/tests/test_vr_combat_routing.cpp")
test(vr_handedness "${R}/tests/test_vr_handedness.cpp")
test(engine_arm_driver "${R}/tests/test_engine_arm_driver.cpp" "${R}/mod/src/EngineArmDriver.cpp" "${R}/mod/src/EngineArmProbe.cpp" "${R}/mod/src/EngineBodyProbe.cpp")
target_link_libraries(engine_arm_driver_tests PRIVATE minhook spdlog)
add_test(NAME engine_arm_driver_config COMMAND engine_arm_driver_tests "${R}/config/index_hands_axis_candidate.calibration" "${R}/config/index_hands_outward_45_candidate.calibration")
test(vr_session "${R}/tests/test_vr_session.cpp" "${R}/mod/src/VrSession.cpp" "${R}/mod/src/EyeTextureCapture.cpp")
target_link_libraries(vr_session_tests PRIVATE d3d12 dxgi)
add_test(NAME vr_session_cpu COMMAND vr_session_tests --cpu-only)
test(vr_gestures "${R}/tests/test_vr_gestures.cpp")
test(swing_melee "${R}/tests/test_swing_melee.cpp")
test(engine_weapon_probe "${R}/tests/test_engine_weapon_probe.cpp" "${R}/mod/src/EngineWeaponProbe.cpp" "${R}/mod/src/EngineBodyProbe.cpp")
target_link_libraries(engine_weapon_probe_tests PRIVATE minhook)
