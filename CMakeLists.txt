cmake_minimum_required(VERSION 3.27)

project(
  sm2vr
  VERSION 1.0.0
  LANGUAGES C CXX
)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

include(FetchContent)

# Static C++ runtime everywhere: the game ships its own, older msvcp140.dll (14.34) next to the exe,
# and a DLL built with a newer toolset must not bind to it.
set(CMAKE_POLICY_DEFAULT_CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_compile_definitions(WIN32_LEAN_AND_MEAN NOMINMAX _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR _ITERATOR_DEBUG_LEVEL=0)
add_compile_options(/EHsc /Zc:__cplusplus /Zc:preprocessor /Zc:enumTypes /Zc:templateScope /MP)

# The game ships Streamline 2.7.30 (sl.interposer.dll file version).
set(STREAMLINE_VERSION "v2.7.30" CACHE STRING "Version of NVIDIA Streamline headers to build against")

# Framework core: refs/vrframework-core is fetched by scripts/fetch-tools.sh (not vendored).
set(VRFRAMEWORK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/refs/vrframework-core" CACHE PATH "vrframework core checkout")
if(NOT EXISTS "${VRFRAMEWORK_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Missing ${VRFRAMEWORK_DIR}; run scripts/fetch-tools.sh")
endif()

set(IMGUI_USER_CONFIG_PATH "${CMAKE_CURRENT_SOURCE_DIR}/mod/imgui/imconfig.hpp")
FetchContent_Declare(vrframework SOURCE_DIR "${VRFRAMEWORK_DIR}")
FetchContent_MakeAvailable(vrframework)

include(cmake/FrameworkCaptureCompatibility.cmake)
include(cmake/FrameworkHookMonitor.cmake)
set(framework_capture_dir "${CMAKE_CURRENT_BINARY_DIR}/framework-capture")
sm2_framework_capture_source("${VRFRAMEWORK_DIR}/src/D3D12Hook.cpp" "${framework_capture_dir}")
# No rehook while the present detour is intact (docs/FREEZE_REHOOK_2026-10-27.md).
sm2_framework_monitor_d3d12("${framework_capture_dir}/D3D12Hook.cpp")
sm2_framework_monitor_source("${VRFRAMEWORK_DIR}/src/Framework.cpp" "${framework_capture_dir}")
get_target_property(framework_sources vr_framework SOURCES)
foreach(framework_patched D3D12Hook.cpp Framework.cpp)
  set(framework_original "${VRFRAMEWORK_DIR}/src/${framework_patched}")
  list(FIND framework_sources "${framework_original}" framework_index)
  if(framework_index LESS 0)
    message(FATAL_ERROR "Framework source ${framework_patched} missing from target")
  endif()
  list(REMOVE_ITEM framework_sources "${framework_original}")
  list(APPEND framework_sources "${framework_capture_dir}/${framework_patched}")
endforeach()
set_property(TARGET vr_framework PROPERTY SOURCES "${framework_sources}")
target_include_directories(vr_framework PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mod/src")

add_library(common INTERFACE)
target_compile_definitions(common INTERFACE USE_DXGI_HOOK HOOK_DX12_FIRST)
target_link_libraries(vr_framework PRIVATE common)

file(GLOB_RECURSE sources CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/mod/src/*.cpp")
file(GLOB_RECURSE headers CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/mod/src/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/mod/src/*.hpp")

add_library(sm2vr SHARED ${headers} ${sources})
target_link_libraries(sm2vr PRIVATE dbghelp) # HangWatchdog's MiniDumpWriteDump
# Sm2Probe.cpp is past the default 2^16 COFF sections (C1128).
if(MSVC)
  target_compile_options(sm2vr PRIVATE /bigobj)
endif()
target_link_libraries(sm2vr PUBLIC common vr_framework::vr_framework)
target_include_directories(sm2vr PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/mod/src")
target_precompile_headers(sm2vr PRIVATE mod/pch/pch.h)

# The game statically imports dxgi.dll!CreateDXGIFactory1, so the mod loads as a dxgi.dll proxy.
set_target_properties(
  sm2vr
  PROPERTIES
    OUTPUT_NAME dxgi
    # The import library must not be called dxgi.lib: the link also takes the SDK's dxgi.lib as input.
    ARCHIVE_OUTPUT_NAME sm2vr
    LINK_FLAGS "/DELAYLOAD:openvr_api.dll /DELAYLOAD:d3d12.dll /DELAYLOAD:D3DCOMPILER_47.dll"
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin"
)
